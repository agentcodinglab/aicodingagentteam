## 2. 五层执行流转图

```plantuml
@startuml flow
!theme plain
skinparam backgroundColor #FAFAFA

start
:用户请求;
partition "① Route 意图路由" {
  :判定类型 Chat/Explain/QuickEdit/Debug/Build;
  :评估深度、写权限、范围;
  if (需要澄清?) then (是)
    :向用户澄清;
  else (否)
  endif
}

partition "② Plan 计划构建" {
  if (Build 类型?) then (是)
    :生成 DAG 任务图;
    :保存 .aicodingagentteam/plan.json;
    if (用户干预?) then (是)
      :/plan 增删调整;
    else (否)
    endif
  else (QuickEdit)
    :跳过计划，直接执行;
  endif
}

partition "③ Schedule 调度执行" {
  fork
    :写角色串行执行\n(前端→后端);
  fork again
    :评审角色并行 A2A\n(PM/架构/QA/安全/DevOps);
  end fork
  :收集 RoleVerdict;
}

partition "④ Verify 自校正验证" {
  :确定性校验\n构建/测试/lint/契约/安全;
  if (阻塞项?) then (是)
    :生成修复方案;
    :指纹快照防循环;
    if (超过重试上限?) then (是)
      :park 暂停任务;
      stop
    else (否)
      :重新调度修复;
    endif
  else (否)
  endif
}

partition "⑤ Finalize 交付产物" {
  :输出对应深度工件;
  :更新项目记忆库;
  :生成审计证据包;
  :输出 proof-pack + scorecard;
}
stop
@enduml
```

## 3. A2A 协作时序图

```plantuml
@startuml a2a
!theme plain
skinparam backgroundColor #FAFAFA

actor "用户" as U
participant "TUI" as T
participant "Coordinator" as C
queue "A2A Bus\n(Redis)" as B
participant "QA Agent" as QA
participant "Architect Agent" as AR
participant "Codex 宿主" as H

U -> T : /run "需求"
T -> C : gRPC RunRequest
C -> C : Route + Plan(DAG)
C -> B : agent.execute(QA, task)
C -> B : agent.execute(Architect, task)
B -> QA : deliver task
B -> AR : deliver task

par QA 执行
  QA -> H : StartSession + SendTask
  H --> QA : Event stream
  QA -> QA : 生成测试用例
  B <-- QA : agent.progress
else Architect 执行
  AR -> H : StartSession + SendTask
  H --> AR : Event stream
  AR -> AR : 生成架构文档
  B <-- AR : agent.progress
end

C <-- B : agent.result(QA Verdict=blocking)
C <-- B : agent.result(Architect Verdict=accept)
C -> C : 汇聚 Verdict 判定
alt 存在 blocking
  C -> T : park 暂停 + 修复方案
  T -> U : 展示阻塞项，等待 /continue
else 全部 accept
  C -> C : Quality-Gate 校验
  C -> T : 交付产物
  T -> U : 显示完成 + scorecard
end
@enduml
```

## 5. 九角色编排模型图

```plantuml
@startuml roles
!theme plain
skinparam backgroundColor #FAFAFA

skinparam rectangle {
  BackgroundColor<<writer>> #E8F5E9
  BackgroundColor<<reviewer>> #FFF3E0
  BackgroundColor<<controller>> #E3F2FD

rectangle "Product Manager\n产品经理" <<reviewer>> as PM
rectangle "Architect\n架构师" <<reviewer>> as AR

rectangle "Backend Engineer\n后端" <<writer>> as BE
rectangle "QA Engineer\n测试" <<reviewer>> as QA
rectangle "Security Engineer\n安全" <<reviewer>> as SEC
rectangle "DevOps\n运维" <<reviewer>> as DO

COORD --> PM : A2A 委派
COORD --> AR : A2A 委派
COORD --> UI : A2A 委派
COORD --> FE : A2A 写角色(串行)
COORD --> BE : A2A 写角色(串行)
COORD --> QA : A2A 委派
COORD --> SEC : A2A 委派
COORD --> DO : A2A 委派

note bottom of FE
  写角色串行约束:
  同一时刻仅一个写角色
```
